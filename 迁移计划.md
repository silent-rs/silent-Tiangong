Tauri 迁移计划：Hetu -> Tauri + React + shadcn/ui背景
当前 tiangong 项目使用 Hetu 框架（保留模式 Rust GUI）实现桌面 UI。由于 Hetu 框架生态不够成熟，决定迁移到 Tauri 框架，配合 Vite + React + shadcn/ui 实现现代化桌面应用。
目标

保留核心业务逻辑：src/core 目录的所有代码完全保留
新增 Tauri 前端：使用 Vite + React + shadcn/ui 重写 UI
Tauri Commands 暴露 API：将 TiangongState 的方法暴露给前端
参考 Codex 样式：深色主题、VS Code 风格布局

src-tauri/src/commands.rs（新建）
rustuse crate::app::TiangongApp;
use tauri::State;

// 获取会话列表
#[tauri::command]
pub fn get_sessions(state: State<TiangongApp>) -> Result<Vec<Session>, String> {
    state.state.lock()
        .map_err(|e| e.to_string())?
        .get_sessions()
        .iter()
        .map(|s| Session::from_core(s))
        .collect::<Result<Vec<_>, _>>()
        .map_err(|e| e.to_string())
}

// 创建新会话
#[tauri::command]
pub fn create_session(state: State<TiangongApp>) -> Result<Session, String> {
    let core_state = state.state.lock().map_err(|e| e.to_string())?;
    let session = core_state.create_session();
    Ok(Session::from_core(&session)?)
}

// 发送消息
#[tauri::command]
pub fn send_message(state: State<TiangongApp>, content: String, window: tauri::Window) -> Result<(), String> {
    let mut core_state = state.state.lock().map_err(|e| e.to_string())?;
    core_state.set_input_draft(&content);
    core_state.send_current_input().map_err(|e| e.to_string())?;

    // 启动后台任务监听事件
    let state_clone = state.inner().clone();
    tauri::async_runtime::spawn(async move {
        // 轮询事件并通过 emit 发送到前端
        loop {
            if let Ok(snapshot) = state_clone.state.lock().map(|s| s.get_run_snapshot().clone()) {
                let _ = window.emit("run_snapshot", snapshot);
            }
            tokio::time::sleep(tokio::time::Duration::from_millis(100)).await;
        }
    });

    Ok(())
}

typescriptimport { invoke } from '@tauri-apps/api/core';
import { listen } from '@tauri-apps/api/event';

// 类型定义
export interface Session {
  id: string;
  title: string;
  created_at: string;
  updated_at: string;
}

export interface Message {
  id: string;
  role: 'user' | 'assistant' | 'system';
  content: string;
  reasoning_content?: string;
  created_at: string;
}

export interface RunSnapshot {
  status: 'idle' | 'planning' | 'executing' | 'responding';
  current_plan?: any;
  messages: Message[];
}

// API 方法
export const api = {
  // 会话管理
  getSessions: () => invoke<Session[]>('get_sessions'),
  createSession: () => invoke<Session>('create_session'),
  switchSession: (id: string) => invoke<void>('switch_session', { sessionId: id }),
  deleteSession: (id: string) => invoke<void>('delete_session', { sessionId: id }),

  // 消息
  sendMessage: (content: string) => invoke<void>('send_message', { content }),
  cancelTurn: () => invoke<void>('cancel_turn'),

  // 状态
  getRunSnapshot: () => invoke<RunSnapshot>('get_run_snapshot'),

  // MCP
  getMcpServers: () => invoke<any[]>('get_mcp_servers'),
  registerMcpServer: (config: any) => invoke<string>('register_mcp_server', { config }),
  removeMcpServer: (name: string) => invoke<string>('remove_mcp_server', { name }),

  // Skill
  getSkills: () => invoke<any[]>('get_skills'),
  installSkill: (path: string) => invoke<string>('install_skill', { path }),
  removeSkill: (id: string) => invoke<string>('remove_skill', { id }),

  // 事件监听
  onRunSnapshot: (callback: (snapshot: RunSnapshot) => void) =>
    listen<RunSnapshot>('run_snapshot', (event) => callback(event.payload)),
};
3.3 状态管理
frontend/src/store/useStore.ts（新建）
typescriptimport { create } from 'zustand';
import { api, Session, Message, RunSnapshot } from '../api/tauri';

interface AppState {
  // 状态
  sessions: Session[];
  activeSessionId: string | null;
  messages: Message[];
  runStatus: RunSnapshot['status'];
  inputContent: string;

  // 操作
  loadSessions: () => Promise<void>;
  createSession: () => Promise<void>;
  switchSession: (id: string) => Promise<void>;
  sendMessage: (content: string) => Promise<void>;
  setInputContent: (content: string) => void;
}

export const useStore = create<AppState>((set, get) => ({
  sessions: [],
  activeSessionId: null,
  messages: [],
  runStatus: 'idle',
  inputContent: '',

  loadSessions: async () => {
    const sessions = await api.getSessions();
    set({ sessions });
  },

  createSession: async () => {
    const session = await api.createSession();
    set(state => ({
      sessions: [...state.sessions, session],
      activeSessionId: session.id,
      messages: [],
    }));
  },

  switchSession: async (id: string) => {
    await api.switchSession(id);
    set({ activeSessionId: id, messages: [] });
  },

  sendMessage: async (content: string) => {
    set({ inputContent: '' });
    await api.sendMessage(content);
    set({ runStatus: 'planning' });
  },

  setInputContent: (content: string) => set({ inputContent: content }),
}));

export function MainApp() {
  const { loadSessions, runStatus } = useStore();

  useEffect(() => {
    loadSessions();

    // 监听运行状态更新
    const unlisten = api.onRunSnapshot((snapshot) => {
      useStore.setState({
        messages: snapshot.messages,
        runStatus: snapshot.status,
      });
    });

    return () => {
      unlisten.then(fn => fn());
    };
  }, []);

        )}

        {/* 消息列表 */}
        <MessageList />

        {/* 输入框 */}
        <MessageInput />
      </div>
    </div>
  );
}

tsximport { useState, KeyboardEvent } from 'react';
import { useStore } from '../store/useStore';
import { Textarea } from './ui/textarea';
import { Button } from './ui/button';
import { Send } from 'lucide-react';

export function MessageInput() {
  const { inputContent, setInputContent, sendMessage, runStatus } = useStore();
  const [isComposing, setIsComposing] = useState(false);

  const handleKeyDown = (e: KeyboardEvent<HTMLTextAreaElement>) => {
    if (e.key === 'Enter' && !e.shiftKey && !isComposing) {
      e.preventDefault();
      handleSend();
    }
  };

  const handleSend = () => {
    if (inputContent.trim() && runStatus === 'idle') {
      sendMessage(inputContent);
    }
  };

  return (
    <div className="p-4 border-t border-[#3C3C3C]">
      <div className="max-w-3xl mx-auto flex gap-2">
        <Textarea
          value={inputContent}
          onChange={e => setInputContent(e.target.value)}
          onKeyDown={handleKeyDown}
          onCompositionStart={() => setIsComposing(true)}
          onCompositionEnd={() => setIsComposing(false)}
          placeholder="输入消息... (Enter 发送, Shift+Enter 换行)"
          className="min-h-[60px] max-h-[200px] bg-[#2D2D30] border-[#3C3C3C] text-white placeholder:text-[#858585]"
          disabled={runStatus !== 'idle'}
        />
        <Button
          onClick={handleSend}
          disabled={!inputContent.trim() || runStatus !== 'idle'}
          className="self-end bg-[#10A37F] hover:bg-[#0D8A6A]"
        >
          <Send className="w-4 h-4" />
        </Button>
      </div>
    </div>
  );
}
